USE hoteldb;

CREATE TABLE tariff (
    tariff_id INT PRIMARY KEY AUTO_INCREMENT,
    tariff_name VARCHAR(50) NOT NULL,
    price_per_night DECIMAL(10,2) NOT NULL
);

CREATE TABLE room_category (
    category_id INT PRIMARY KEY AUTO_INCREMENT,
    category_name VARCHAR(60) NOT NULL
);

CREATE TABLE room_status (
    room_status_id INT PRIMARY KEY AUTO_INCREMENT,
    status_name VARCHAR(30) NOT NULL
);

CREATE TABLE equipment (
    equipment_id INT PRIMARY KEY AUTO_INCREMENT,
    equipment_name VARCHAR(50) NOT NULL
);

CREATE TABLE guest (
    guest_id INT PRIMARY KEY AUTO_INCREMENT,
    full_name VARCHAR(150) NOT NULL,
    birth_date DATE,
    citizenship VARCHAR(60),
    id_document VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    email VARCHAR(100)
);

CREATE TABLE booking_source (
    source_id INT PRIMARY KEY AUTO_INCREMENT,
    source_name VARCHAR(60) NOT NULL
);

CREATE TABLE booking_status (
    booking_status_id INT PRIMARY KEY AUTO_INCREMENT,
    status_name VARCHAR(30) NOT NULL
);

CREATE TABLE room (
    room_id INT PRIMARY KEY AUTO_INCREMENT,
    tariff_id INT NOT NULL,
    category_id INT NOT NULL,
    room_status_id INT NOT NULL,
    room_name VARCHAR(50),
    rooms_count INT DEFAULT 1,
    area DECIMAL(5,1),
    floor INT,
    building_number INT,
    FOREIGN KEY (tariff_id) REFERENCES tariff(tariff_id),
    FOREIGN KEY (category_id) REFERENCES room_category(category_id),
    FOREIGN KEY (room_status_id) REFERENCES room_status(room_status_id)
);

CREATE TABLE room_equipment (
    room_id INT,
    equipment_id INT,
    PRIMARY KEY (room_id, equipment_id),
    FOREIGN KEY (room_id) REFERENCES room(room_id) ON DELETE CASCADE,
    FOREIGN KEY (equipment_id) REFERENCES equipment(equipment_id) ON DELETE CASCADE
);

CREATE TABLE booking (
    booking_id INT PRIMARY KEY AUTO_INCREMENT,
    guest_id INT NOT NULL,
    category_id INT NOT NULL,
    source_id INT NOT NULL,
    booking_status_id INT NOT NULL,
    check_in_planned DATETIME NOT NULL,
    check_out_planned DATETIME NOT NULL,
    guests_count INT DEFAULT 1,
    special_requests VARCHAR(255),
    FOREIGN KEY (guest_id) REFERENCES guest(guest_id),
    FOREIGN KEY (category_id) REFERENCES room_category(category_id),
    FOREIGN KEY (source_id) REFERENCES booking_source(source_id),
    FOREIGN KEY (booking_status_id) REFERENCES booking_status(booking_status_id)
);

CREATE TABLE stay (
    stay_id INT PRIMARY KEY AUTO_INCREMENT,
    booking_id INT NOT NULL,
    room_id INT NOT NULL,
    check_in_actual DATETIME,
    check_out_actual DATETIME,
    FOREIGN KEY (booking_id) REFERENCES booking(booking_id),
    FOREIGN KEY (room_id) REFERENCES room(room_id)
);

SELECT g.full_name, b.check_in_planned, bs.status_name
FROM guest g, booking b, booking_status bs
WHERE g.guest_id = b.guest_id 
  AND b.booking_status_id = bs.booking_status_id 
  AND bs.status_name = 'Предварительное'
ORDER BY b.check_in_planned ASC;

SELECT r.room_name, rc.category_name, t.tariff_name, t.price_per_night
FROM room r, room_category rc, tariff t
WHERE r.category_id = rc.category_id 
  AND r.tariff_id = t.tariff_id
  AND t.price_per_night > 4000
ORDER BY t.price_per_night DESC;

FROM guest g, booking b, booking_source s
WHERE g.guest_id = b.guest_id 
  AND b.source_id = s.source_id
  AND s.source_name = 'Сайт'
ORDER BY b.guests_count DESC;

USE hoteldb;

CREATE TABLE IF NOT EXISTS action_log (
    log_id INT PRIMARY KEY AUTO_INCREMENT,
    action_message VARCHAR(255),
    action_date DATETIME
);

DELIMITER //

CREATE PROCEDURE GetTariffStats()
BEGIN
    SELECT 
        SUM(price_per_night) AS total_sum,
        AVG(price_per_night) AS average_price,
        MAX(price_per_night) AS max_price,
        MIN(price_per_night) AS min_price
    FROM tariff;
END //

DELIMITER ;

DELIMITER //

CREATE PROCEDURE AddLogEntry()
BEGIN
    INSERT INTO action_log (action_message, action_date) 
    VALUES ('Выполнен расчет статистики по тарифам', NOW());
END //

DELIMITER ;

DELIMITER //

CREATE FUNCTION CountGuestsFromCountry(country_name VARCHAR(60)) 
RETURNS INT
DETERMINISTIC
BEGIN

    DECLARE guest_count INT;   
 
    SELECT COUNT(*) INTO guest_count 
    FROM guest 
    WHERE citizenship = country_name;
    
    RETURN guest_count;
END //

DELIMITER ;

DELIMITER //

CREATE FUNCTION CountRoomsOnFloor(floor_num INT) 
RETURNS INT
DETERMINISTIC
BEGIN

    DECLARE room_count INT; 

    SELECT COUNT(*) INTO room_count 
    FROM room 
    WHERE floor = floor_num;
    
    RETURN room_count;
END //

DELIMITER ;
